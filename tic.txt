import pandas as pd
from IPython.display import display

# ---- Pure-Python ROUGE-L F1 (token-level LCS) ----
def rougeL_f1(a: str, b: str) -> float:
    """Compute a lightweight ROUGE-L F1 using token-level longest common subsequence."""
    xa, xb = a.lower().split(), b.lower().split()
    m, n = len(xa), len(xb)
    if m == 0 or n == 0:
        return 0.0
    # LCS DP
    dp = [[0]*(n+1) for _ in range(m+1)]
    for i in range(1, m+1):
        ai = xa[i-1]
        for j in range(1, n+1):
            if ai == xb[j-1]:
                dp[i][j] = dp[i-1][j-1] + 1
            else:
                dp[i][j] = dp[i-1][j] if dp[i-1][j] >= dp[i][j-1] else dp[i][j-1]
    L = dp[m][n]
    p = L / float(n)
    r = L / float(m)
    return (2*p*r) / (p + r + 1e-12)

def precision_recall_at_k(ref_text, retrieved_texts, k, thresh=0.50):
    """
    Count a retrieved doc as relevant if rougeL_f1(ref_context, doc) >= `thresh`.
    Returns (precision@k, recall@k). We treat the reference context as a single gold.
    """
    rel = 0
    for t in retrieved_texts[:k]:
        if rougeL_f1(ref_text, t) >= thresh:
            rel += 1
    precision = rel / max(len(retrieved_texts[:k]), 1)
    recall = rel / 1.0
    return precision, recall

def evaluate_validation(df_val, strategy="hybrid", overlap_halluc_thresh=0.10, progress_every=5):
    """
    Runs retrieval+generation over the validation set and computes:
      - precision@k, recall@k  (retriever quality proxy)
      - rougeL_answer_vs_ref   (answer relevance)
      - context_overlap        (answer vs retrieved text)
      - hallucination_flag     (low overlap => True)
    Requires globals: retrieve_docs, generate_answer, get_llm, RETRIEVAL_K
    """
    print(f"[evaluate_validation] strategy={strategy}", flush=True)

    if df_val is None or df_val.empty:
        print("⚠️ No validation rows; returning None.", flush=True)
        return None

    # sanity check required globals
    missing = [name for name in ("retrieve_docs", "generate_answer", "get_llm", "RETRIEVAL_K") if name not in globals()]
    if missing:
        raise RuntimeError(f"Missing required globals: {missing}. Run earlier cells to define them.")

    llm = get_llm()
    rows = []
    total = len(df_val)

    for i, row in enumerate(df_val.itertuples(index=False), start=1):
        q = str(getattr(row, "question"))
        ref_ctx = str(getattr(row, "reference_context"))
        ref_ans = str(getattr(row, "reference_answer"))

        # Retrieval
        docs = retrieve_docs(q, strategy=strategy)
        texts = [getattr(d, "page_content", str(d)) for d in docs]

        # Generation (uncertainty handling)
        if texts:
            gen = generate_answer(q, docs, llm=llm)
        else:
            gen = "The question cannot be answered using the available documents."

        # Metrics
        p, r = precision_recall_at_k(ref_ctx, texts, k=RETRIEVAL_K)
        rougeL = rougeL_f1(ref_ans, gen)
        ctx_overlap = rougeL_f1(" ".join(texts), gen) if texts else 0.0
        hallucinated = ctx_overlap < overlap_halluc_thresh

        rows.append({
            "question": q,
            "precision@k": p,
            "recall@k": r,
            "rougeL_answer_vs_ref": rougeL,
            "context_overlap": ctx_overlap,
            "hallucination_flag": hallucinated,
            "n_ctx": len(texts),
        })

        if progress_every and i % progress_every == 0:
            print(f"  processed {i}/{total}", flush=True)

    out = pd.DataFrame(rows)
    print(f"[evaluate_validation] done: {out.shape[0]} rows", flush=True)
    return out

def run_validation_benchmark(df_val):
    """Convenience helper to run both strategies and show summaries."""
    print("=== Running semantic ===", flush=True)
    report_sem = evaluate_validation(df_val, strategy="semantic")
    print("=== Running hybrid ===", flush=True)
    report_hyb = evaluate_validation(df_val, strategy="hybrid")

    def _show(df, name):
        print(f"\n--- {name} ---", flush=True)
        if df is None:
            print("None (likely missing validation file)."); return
        if getattr(df, "empty", True):
            print("Empty DataFrame (0 rows)"); display(df); return
        print("shape:", df.shape)
        display(df.head(10))
        print("describe():")
        display(df.describe(include="all"))
        # quick means
        numeric_cols = ["precision@k","recall@k","rougeL_answer_vs_ref","context_overlap"]
        means = {c: float(df[c].mean()) for c in numeric_cols if c in df.columns}
        print("means:", means)

    _show(report_sem, "SEMANTIC")
    _show(report_hyb, "HYBRID")
    return report_sem, report_hyb
